cmake_minimum_required(VERSION 3.25)
project(synctest C CXX)
set(CMAKE_CXX_STANDARD 23)
set(qt "${WEBBY_QT_ENV}/opt/qt6-wasi/wasi")
add_executable(synctest test.cpp ${WASI_QT_COMPAT_C} ${CMAKE_CURRENT_SOURCE_DIR}/../../ui/SyncDoc.cpp ${CMAKE_CURRENT_SOURCE_DIR}/../../ui/SyncKeys.cpp)
target_include_directories(synctest PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../../ui ${WEBBY_WASI_SYSROOT}/include "${qt}/include" "${qt}/include/QtCore" "${qt}/mkspecs/wasi-clang")
target_compile_definitions(synctest PRIVATE QT_CORE_LIB QT_STATIC QT_NO_KEYWORDS)
set_source_files_properties(${WASI_QT_COMPAT_C} PROPERTIES COMPILE_OPTIONS "-w")
target_compile_options(synctest PRIVATE -Wall -Wextra -Werror -Wno-padded-bitfield -Wmissing-field-initializers -Wshadow)
target_link_libraries(synctest PRIVATE "${qt}/lib/libQt6Core.a" "${qt}/lib/libQt6BundledPcre2.a" "${qt}/lib/libQt6BundledZLIB.a" ${WEBBY_WASI_SYSROOT}/lib/libcrypto.a)
